cmake_minimum_required(VERSION 3.16)
project(paperdiy_reader_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SANITIZE "Enable address and undefined behavior sanitizers" ON)
if(SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-UNDEBUG)
include_directories(src)
find_package(OpenSSL REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(M5GFX_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.pio/libdeps/paperdiy/M5GFX" CACHE PATH "M5GFX source directory")
add_executable(controls tests/controls.cpp)
add_executable(power_policy tests/power_policy.cpp)
add_executable(english_layout tests/english_layout.cpp)
add_executable(cache_adapter tests/cache_adapter.cpp)
target_link_libraries(cache_adapter PRIVATE OpenSSL::Crypto)
target_compile_options(cache_adapter PRIVATE -Wno-deprecated-declarations)
add_executable(reading_state tests/reading_state.cpp)
target_link_libraries(reading_state PRIVATE OpenSSL::Crypto)
target_compile_options(reading_state PRIVATE -Wno-deprecated-declarations)
add_executable(font_metrics tests/font_metrics.cpp)
add_executable(reader_integration tests/reader_integration.cpp)
target_include_directories(reader_integration BEFORE PRIVATE tests/stubs)
target_link_libraries(reader_integration PRIVATE OpenSSL::Crypto)
target_compile_options(reader_integration PRIVATE -Wno-deprecated-declarations)
add_executable(ui_simulation tests/ui_simulation.cpp src/ui.cpp src/reader.cpp src/display_driver.cpp)
target_include_directories(ui_simulation BEFORE PRIVATE tests/ui_stubs tests/stubs)
target_link_libraries(ui_simulation PRIVATE OpenSSL::Crypto)
target_compile_options(ui_simulation PRIVATE -Wno-deprecated-declarations)
add_library(miniz STATIC "${M5GFX_DIR}/src/lgfx/utility/lgfx_miniz.c")
add_executable(epub_core tests/epub_core.cpp lib/tinyxml2/tinyxml2.cpp)
target_include_directories(epub_core PRIVATE lib/tinyxml2 "${M5GFX_DIR}/src")
target_link_libraries(epub_core PRIVATE miniz)
enable_testing()
add_test(NAME controls COMMAND controls)
add_test(NAME power_policy COMMAND power_policy)
add_test(NAME english_layout COMMAND english_layout)
add_test(NAME hyphenation_data COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/build_hyphenation.py" --check)
add_test(NAME cache_adapter COMMAND cache_adapter)
add_test(NAME reading_state COMMAND reading_state)
add_test(NAME font_metrics COMMAND font_metrics)
add_test(NAME reader_integration COMMAND reader_integration)
add_test(NAME ui_simulation COMMAND ui_simulation)
add_test(NAME font_generation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/font_generation.py")
add_test(NAME epub_import COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/epub_fixtures.py" "$<TARGET_FILE:epub_core>")
